-- 查看某只股票的所有日线数据（最近 10 条）
SELECT * FROM daily_raw 
WHERE ts_code = '002948.SZ' 
ORDER BY trade_date DESC 
LIMIT 10;
另外一个总体信息表格
SELECT * FROM daily_basic WHERE ts_code = '002948.SZ' ORDER BY trade_date DESC LIMIT 10;

-- 查看某只股票的复权因子
SELECT * FROM adjust_factor 
WHERE ts_code = '601187.SH' 
ORDER BY trade_date DESC;

-- 查看各股票的数据同步状态
SELECT * FROM data_meta;

-- 统计每只股票的数据条数
SELECT ts_code, COUNT(*) AS record_count 
FROM daily_raw 
GROUP BY ts_code;

-- 1. 直接查看表中有哪些股票代码（前10个）
SELECT DISTINCT ts_code FROM daily_raw LIMIT 10;

-- 2. 使用 TRIM 去除可能的前后空格
SELECT * FROM daily_raw 
WHERE TRIM(ts_code) = '002948.SZ' 
ORDER BY trade_date DESC 
LIMIT 10;

-- 3. 使用 LIKE 模糊匹配（不推荐，仅用于排查）
SELECT * FROM daily_raw 
WHERE ts_code LIKE '%002948%' 
ORDER BY trade_date DESC 
LIMIT 10;

-- 4. 直接查看全表数据（如果数据量不大）
SELECT * FROM daily_raw LIMIT 10;


信息项
ts_code：股票代码

trade_date：交易日期

open：开盘价

high：最高价

low：最低价

close：收盘价

pre_close：前收盘价

change：涨跌额（收盘-前收）

pct_chg：涨跌幅（%）

vol：成交量（手）

amount：成交金额（元）

total_mv：总市值（元）

circ_mv：流通市值（元）

turnover_rate：换手率（%）

turnover_rate_f：自由流通换手率（%）

volume_ratio：量比

pe：市盈率（静态）

pe_ttm：滚动市盈率（最近12个月）

pb：市净率
